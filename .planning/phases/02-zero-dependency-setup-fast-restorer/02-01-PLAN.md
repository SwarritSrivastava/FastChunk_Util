       ```rust
       pub struct ChunkedReader {
           part_paths: Vec<PathBuf>,
           current_idx: usize,
           current_file: Option<File>,
       }

       impl ChunkedReader {
           pub fn new(part_paths: Vec<PathBuf>) -> io::Result<Self> {
               let mut reader = Self {
                   part_paths,
                   current_idx: 0,
                   current_file: None,
               };
               reader.open_current()?;
               Ok(reader)
           }

           fn open_current(&mut self) -> io::Result<()> {
               if self.current_idx < self.part_paths.len() {
                   let path = &self.part_paths[self.current_idx];
                   let file = File::open(path)?;
                   self.current_file = Some(file);
               } else {
                   self.current_file = None;
               }
               Ok(())
           }
       }
       ```
    2. Implement `std::io::Read` for `ChunkedReader`:
       ```rust
       impl std::io::Read for ChunkedReader {
           fn read(&mut self, buf: &mut [u8]) -> io::Result<usize> {
               if buf.is_empty() {
                   return Ok(0);
               }

               loop {
                   let file = match &mut self.current_file {
                       Some(f) => f,
                       None => return Ok(0), // EOF on all parts
                   };

                   let bytes_read = file.read(buf)?;
                   if bytes_read > 0 {
                       return Ok(bytes_read);
                   }

                   // Current file reached EOF, advance to next part
                   self.current_idx += 1;
                   self.open_current()?;
               }
           }
       }
       ```
